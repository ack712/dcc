def mod_inverse_iterative(a, m):
    """Finds the modular inverse of a modulo m using iterative EEA"""
    m0 = m
    y = 0
    x = 1

    if m == 1:
        return 0

    while a > 1:
        # q is quotient
        q = a // m
        t = m

        # m is remainder now, process same as Euclid's algo
        m = a % m
        a = t
        t = y

        # Update x and y
        y = x - q * y
        x = t

    # Make x positive
    if x < 0:
        x = x + m0

    # If gcd(a, m) is not 1, inverse doesn't exist
    # Note: 'a' contains the last non-zero remainder (the GCD)
    if a != 1:
        raise ValueError("Inverse does not exist")

    return x

# Example Usage:
# Find the inverse of 3 modulo 11
print(mod_inverse_iterative(3, 11))  # Output: 4
