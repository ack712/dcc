from scipy import fft
import numpy as np
import cv2
import matplotlib.pyplot as plt

# Load the uploaded image as grayscale
# Using the local file that was uploaded in previous steps
img_path = 'oskar-smethurst-B1GtwanCbiw-unsplash.jpg'
img = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE)

if img is None:
    print(f"Error: Could not read image from {img_path}. Please make sure the file exists.")
else:
    # Convert the grayscale image to float32
    imf = np.float32(img)

    # Find discrete cosine transform (2D DCT)
    dst1 = fft.dctn(imf)

    # Zero out high-frequency coefficients (compression)
    h, w = dst1.shape
    # Clamp the loop indices to prevent IndexError
    for i in range(min(100, h), min(255, h)):
        for j in range(min(100, w), min(255, w)):
            dst1[i, j] = 0

    # Apply inverse 2D discrete cosine transform (IDCT)
    img1 = fft.idctn(dst1)

    # Plot the results
    plt.figure(figsize=(10, 5))
    plt.gray()
    plt.subplot(121)
    plt.imshow(img)
    plt.axis('off')
    plt.title('Original Image', size=12)

    plt.subplot(122)
    plt.imshow(img1)
    plt.axis('off')
    plt.title('Reconstructed Image (DCT+IDCT)', size=12)
    plt.show()
